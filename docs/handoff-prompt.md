```

# 2. Create tasks
TaskCreate: subject="...", description="...", activeForm="..."

# 3. Spawn agents (ALWAYS use isolation: "worktree")

  subagent_type: "general-purpose"

  run_in_background: true
  prompt: '...'

SendMessage: type="shutdown_request", recipient="agent-name"
```
